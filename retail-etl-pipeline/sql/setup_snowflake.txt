CREATE OR REPLACE TASK LOAD_SALES_CLEAN_TASK
    WAREHOUSE = RETAIL_WH
    SCHEDULE = 'USING CRON 0 6 * * * UTC'
AS
BEGIN
    TRUNCATE TABLE RETAIL_DB.CLEANSED.SALES_CLEAN;
    COPY INTO RETAIL_DB.CLEANSED.SALES_CLEAN
        FROM @RETAIL_DB.CLEANSED.S3_PROCESSED_STAGE
        FILES = ('sales_clean.csv')
        FORCE = TRUE
        ON_ERROR = 'ABORT_STATEMENT';
END;

ALTER TASK LOAD_SALES_CLEAN_TASK RESUME;

EXECUTE TASK LOAD_SALES_CLEAN_TASK;

SELECT COUNT(*) FROM SALES_CLEAN;

SELECT name, state, error_message, completed_time
FROM TABLE(INFORMATION_SCHEMA.TASK_HISTORY(TASK_NAME => 'LOAD_SALES_CLEAN_TASK'))
ORDER BY scheduled_time DESC
LIMIT 5;




USE ROLE RETAIL_ROLE;
USE WAREHOUSE RETAIL_WH;
USE SCHEMA RETAIL_DB.STAR;

CREATE OR REPLACE TABLE DIM_DATE (
    DATE_KEY     NUMBER(8,0)  NOT NULL PRIMARY KEY,
    FULL_DATE    DATE         NOT NULL,
    YEAR         NUMBER(4,0)  NOT NULL,
    QUARTER      NUMBER(1,0)  NOT NULL,
    MONTH        NUMBER(2,0)  NOT NULL,
    MONTH_NAME   VARCHAR(3)   NOT NULL,
    DAY          NUMBER(2,0)  NOT NULL,
    DAY_OF_WEEK  NUMBER(1,0)  NOT NULL,
    DAY_NAME     VARCHAR(3)   NOT NULL,
    IS_WEEKEND   BOOLEAN      NOT NULL
);

CREATE OR REPLACE TABLE DIM_PRODUCT (
    PRODUCT_ID   NUMBER(10,0) NOT NULL PRIMARY KEY,
    CATEGORY     VARCHAR(50)  NOT NULL,
    BRAND        VARCHAR(50)  NOT NULL,
    RATING       NUMBER(2,1),
    IN_STOCK     BOOLEAN,
    LAUNCH_DATE  DATE
);

CREATE OR REPLACE TABLE FACT_SALES (
    SALES_ID        NUMBER(10,0)  NOT NULL PRIMARY KEY,
    DATE_KEY        NUMBER(8,0)   NOT NULL REFERENCES DIM_DATE (DATE_KEY),
    PRODUCT_ID      NUMBER(10,0)  NOT NULL REFERENCES DIM_PRODUCT (PRODUCT_ID),
    SALE_TIMESTAMP  TIMESTAMP_NTZ NOT NULL,
    SALES_MONTH     DATE          NOT NULL,
    REGION          VARCHAR(20)   NOT NULL,
    CATEGORY        VARCHAR(50)   NOT NULL,
    ORDER_STATUS    VARCHAR(20)   NOT NULL,
    QUANTITY        NUMBER(5,0)   NOT NULL,
    PRICE           NUMBER(10,2)  NOT NULL,
    DISCOUNT        NUMBER(4,2)   NOT NULL,
    REVENUE         NUMBER(12,2)  NOT NULL
);

INSERT OVERWRITE INTO DIM_DATE
SELECT DISTINCT
    TO_NUMBER(TO_CHAR(SALE_TIMESTAMP::DATE, 'YYYYMMDD')),
    SALE_TIMESTAMP::DATE,
    YEAR(SALE_TIMESTAMP),
    QUARTER(SALE_TIMESTAMP),
    MONTH(SALE_TIMESTAMP),
    MONTHNAME(SALE_TIMESTAMP),
    DAY(SALE_TIMESTAMP),
    DAYOFWEEKISO(SALE_TIMESTAMP),
    DAYNAME(SALE_TIMESTAMP),
    DAYOFWEEKISO(SALE_TIMESTAMP) IN (6, 7)
FROM RETAIL_DB.CLEANSED.SALES_CLEAN;

INSERT OVERWRITE INTO DIM_PRODUCT
SELECT DISTINCT PRODUCT_ID, CATEGORY, BRAND, RATING, IN_STOCK, LAUNCH_DATE
FROM RETAIL_DB.CLEANSED.SALES_CLEAN;

INSERT OVERWRITE INTO FACT_SALES
SELECT
    SALES_ID,
    TO_NUMBER(TO_CHAR(SALE_TIMESTAMP::DATE, 'YYYYMMDD')),
    PRODUCT_ID,
    SALE_TIMESTAMP,
    DATE_TRUNC('MONTH', SALE_TIMESTAMP)::DATE,
    REGION,
    CATEGORY,
    ORDER_STATUS,
    QUANTITY,
    PRICE,
    DISCOUNT,
    REVENUE
FROM RETAIL_DB.CLEANSED.SALES_CLEAN;


SELECT 'DIM_DATE' AS table_name, COUNT(*) AS row_count FROM DIM_DATE
UNION ALL SELECT 'DIM_PRODUCT', COUNT(*) FROM DIM_PRODUCT
UNION ALL SELECT 'FACT_SALES', COUNT(*) FROM FACT_SALES;

CREATE OR REPLACE MATERIALIZED VIEW MV_SALES_BY_REGION_MONTH AS
SELECT
    REGION,
    SALES_MONTH,
    COUNT(*)       AS ORDER_COUNT,
    SUM(QUANTITY)  AS TOTAL_QUANTITY,
    SUM(REVENUE)   AS TOTAL_REVENUE
FROM RETAIL_DB.STAR.FACT_SALES
GROUP BY REGION, SALES_MONTH;

CREATE OR REPLACE MATERIALIZED VIEW MV_TOP_PRODUCTS_BY_REVENUE AS
SELECT
    PRODUCT_ID,
    CATEGORY,
    COUNT(*)       AS ORDER_COUNT,
    SUM(QUANTITY)  AS TOTAL_QUANTITY,
    SUM(REVENUE)   AS TOTAL_REVENUE
FROM RETAIL_DB.STAR.FACT_SALES
GROUP BY PRODUCT_ID, CATEGORY;

CREATE OR REPLACE MATERIALIZED VIEW MV_REVENUE_TREND AS
SELECT
    SALE_TIMESTAMP::DATE AS SALES_DATE,
    COUNT(*)             AS ORDER_COUNT,
    SUM(REVENUE)         AS TOTAL_REVENUE
FROM RETAIL_DB.STAR.FACT_SALES
GROUP BY SALE_TIMESTAMP::DATE;

CREATE OR REPLACE MATERIALIZED VIEW MV_CATEGORY_PERFORMANCE AS
SELECT
    CATEGORY,
    COUNT(*)       AS ORDER_COUNT,
    SUM(QUANTITY)  AS TOTAL_QUANTITY,
    SUM(REVENUE)   AS TOTAL_REVENUE,
    AVG(DISCOUNT)  AS AVG_DISCOUNT
FROM RETAIL_DB.STAR.FACT_SALES
GROUP BY CATEGORY;

CREATE OR REPLACE MATERIALIZED VIEW MV_HOURLY_SALES AS
SELECT
    REGION,
    CATEGORY,
    HOUR(SALE_TIMESTAMP) AS SALES_HOUR,
    COUNT(*)             AS ORDER_COUNT,
    SUM(REVENUE)         AS TOTAL_REVENUE
FROM RETAIL_DB.STAR.FACT_SALES
GROUP BY REGION, CATEGORY, HOUR(SALE_TIMESTAMP);

CREATE OR REPLACE VIEW V_REVENUE_CONCENTRATION AS
WITH region_revenue AS (
    SELECT REGION, SUM(REVENUE) AS REGION_REVENUE
    FROM RETAIL_DB.STAR.FACT_SALES
    GROUP BY REGION
)
SELECT
    REGION,
    REGION_REVENUE,
    ROUND(REGION_REVENUE / SUM(REGION_REVENUE) OVER (), 4) AS REVENUE_SHARE,
    ROUND(
        SUM(REGION_REVENUE) OVER (ORDER BY REGION_REVENUE DESC ROWS BETWEEN UNBOUNDED PRECEDING AND CURRENT ROW)
        / SUM(REGION_REVENUE) OVER (), 4
    ) AS CUMULATIVE_SHARE
FROM region_revenue;


SELECT *
FROM MV_HOURLY_SALES
QUALIFY ROW_NUMBER() OVER (PARTITION BY REGION, CATEGORY ORDER BY TOTAL_REVENUE DESC) = 1
ORDER BY REGION, CATEGORY;

SELECT * FROM V_REVENUE_CONCENTRATION ORDER BY REGION_REVENUE DESC;
